## Interfaces (dùng chung — `src/lib/ekid/ekid-system.ts`)
```ts
export type KidRules = { dailyLimitMin: number; allowedFrom: string|null; allowedTo: string|null; maxLessonsPerDay: number; requirePassToAdvance: boolean };
export type KidRewards = { pointsPerPassedLesson: number; badgeMilestones: number[]; custom: { label: string; points: number }[] };
export type KidSettings = { rules: KidRules; subjects: string[]; rewards: KidRewards };
export const DEFAULT_SETTINGS: KidSettings;                    // rules mặc định, subjects [] = tất cả, milestones [3,5,10]
export function normalizeKidSettings(raw: unknown): KidSettings; // merge default + raw (an toàn)
export function isWithinAllowedHours(settings: KidSettings, now: Date): boolean;
```
`kid.settings.subjects` rỗng ⇒ không lọc theo môn (học tất cả nhóm của lứa).
